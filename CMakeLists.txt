cmake_minimum_required(VERSION 3.24)
set(FAST_PALETTE_VERSION "1.3.0" CACHE STRING "Release version")
project(FastPalette VERSION ${FAST_PALETTE_VERSION} LANGUAGES CXX RC)
configure_file(resources/app.rc "${CMAKE_BINARY_DIR}/generated/app.rc" @ONLY)
configure_file(resources/app.manifest "${CMAKE_BINARY_DIR}/generated/app.manifest" @ONLY)
if(NOT WIN32)
  message(FATAL_ERROR "Fast Palette is built exclusively for Windows.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:/W4;/permissive-;/utf-8;/EHsc>")
add_library(palette_core src/calculator.cpp src/search.cpp src/conversions.cpp)
target_include_directories(palette_core PUBLIC src)
add_library(palette_keyboard src/keyboard.cpp)
target_include_directories(palette_keyboard PUBLIC src)
target_link_libraries(palette_keyboard PUBLIC user32)
add_library(palette_platform src/catalog.cpp src/settings.cpp src/providers.cpp src/modules.cpp)
target_include_directories(palette_platform PUBLIC src)
target_link_libraries(palette_platform PUBLIC palette_core shell32 ole32 propsys advapi32 user32 uuid)
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE palette_core)
add_test(NAME core COMMAND core_tests)
add_executable(conversions_tests tests/conversions_tests.cpp)
target_link_libraries(conversions_tests PRIVATE palette_core)
add_test(NAME conversions COMMAND conversions_tests)
add_executable(result_actions_tests tests/result_actions_tests.cpp)
target_link_libraries(result_actions_tests PRIVATE palette_platform)
add_test(NAME result_actions COMMAND result_actions_tests)
add_executable(modules_tests tests/modules_tests.cpp)
target_link_libraries(modules_tests PRIVATE palette_platform)
add_test(NAME modules COMMAND modules_tests)
add_executable(keyboard_tests tests/keyboard_tests.cpp)
target_link_libraries(keyboard_tests PRIVATE palette_keyboard)
add_test(NAME keyboard COMMAND keyboard_tests)
add_executable(platform_tests tests/platform_tests.cpp)
add_executable(launch_probe WIN32 tests/launch_probe.cpp)
target_link_libraries(launch_probe PRIVATE shell32)
add_dependencies(platform_tests launch_probe)
target_link_libraries(platform_tests PRIVATE palette_platform)
add_test(NAME platform COMMAND platform_tests)
add_executable(providers_tests tests/providers_tests.cpp)
target_link_libraries(providers_tests PRIVATE palette_platform)
add_test(NAME providers COMMAND providers_tests)
add_executable(shortcut_capture_tests tests/shortcut_capture_tests.cpp)
target_include_directories(shortcut_capture_tests PRIVATE src)
add_test(NAME shortcut_capture COMMAND shortcut_capture_tests)
add_executable(palette_benchmark tests/benchmark.cpp)
target_link_libraries(palette_benchmark PRIVATE palette_core)
list(APPEND CMAKE_PREFIX_PATH "${CMAKE_SOURCE_DIR}/build/deps/qt-static")
find_package(Qt6 6.11 REQUIRED COMPONENTS Widgets Test)
get_target_property(PALETTE_QT_CORE_TYPE Qt6::Core TYPE)
if(NOT PALETTE_QT_CORE_TYPE STREQUAL "STATIC_LIBRARY")
  message(FATAL_ERROR "Fast Palette distribution requires a static Qt build.")
endif()
add_library(palette_ui src/window.cpp src/settings_dialog.cpp src/ui_theme.cpp src/updater.cpp src/number_format.cpp)
qt_add_resources(palette_ui "theme_assets" PREFIX "/icons" BASE "resources"
  FILES resources/check-dark.png resources/check-light.png)
target_include_directories(palette_ui PUBLIC src)
target_compile_definitions(palette_ui PUBLIC FAST_PALETTE_VERSION="${PROJECT_VERSION}")
target_link_libraries(palette_ui PUBLIC palette_platform palette_keyboard Qt6::Widgets wtsapi32 winhttp version)
qt_add_executable(FastPalette WIN32 src/main.cpp "${CMAKE_BINARY_DIR}/generated/app.rc" "${CMAKE_BINARY_DIR}/generated/app.manifest")
qt_add_resources(FastPalette "licenses" PREFIX "/licenses" BASE "resources"
  FILES resources/THIRD_PARTY_NOTICES.txt)
target_include_directories(FastPalette PRIVATE resources)
target_link_libraries(FastPalette PRIVATE palette_ui)
qt_import_plugins(FastPalette INCLUDE Qt6::QWindowsIntegrationPlugin)
qt_add_executable(qt_ui_tests tests/qt_ui_tests.cpp)
target_link_libraries(qt_ui_tests PRIVATE palette_ui Qt6::Test)
qt_import_plugins(qt_ui_tests INCLUDE Qt6::QWindowsIntegrationPlugin)
add_test(NAME qt_ui COMMAND qt_ui_tests)
qt_add_executable(qt_settings_tests tests/qt_settings_tests.cpp)
target_link_libraries(qt_settings_tests PRIVATE palette_ui Qt6::Test)
qt_import_plugins(qt_settings_tests INCLUDE Qt6::QWindowsIntegrationPlugin)
add_test(NAME qt_settings COMMAND qt_settings_tests)
add_executable(updater_tests tests/updater_tests.cpp "${CMAKE_BINARY_DIR}/generated/app.rc")
target_include_directories(updater_tests PRIVATE resources)
target_link_libraries(updater_tests PRIVATE palette_ui)
add_test(NAME updater COMMAND updater_tests)
add_executable(number_format_tests tests/number_format_tests.cpp)
target_link_libraries(number_format_tests PRIVATE palette_ui)
add_test(NAME number_format COMMAND number_format_tests)
